    27→// ---------------- Cover layout helpers (from design-system) ----------------
    28→function splitTitleLines(title, charsPerLine) {
    29→  if (title.length <= charsPerLine) return [title];
    30→  const breakAfter = new Set([..."，。、；：！？", ..."的与和及之在于为", ..."-_—–·/", ..." \t"]);
    31→  const lines = [];
    32→  let remaining = title;
    33→  while (remaining.length > charsPerLine) {
    34→    let breakAt = -1;
    35→    for (let i = charsPerLine; i >= Math.floor(charsPerLine * 0.6); i--) {
    36→      if (i < remaining.length && breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
    37→    }
    38→    if (breakAt === -1) {
    39→      const limit = Math.min(remaining.length, Math.ceil(charsPerLine * 1.3));
    40→      for (let i = charsPerLine + 1; i < limit; i++) {
    41→        if (breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
    42→      }
    43→    }
    44→    if (breakAt === -1) breakAt = charsPerLine;
    45→    lines.push(remaining.slice(0, breakAt).trim());
    46→    remaining = remaining.slice(breakAt).trim();
    47→  }
    48→  if (remaining) lines.push(remaining);
    49→  if (lines.length > 1 && lines[lines.length - 1].length <= 2) {
    50→    const last = lines.pop();
    51→    lines[lines.length - 1] += last;
    52→  }
    53→  return lines;
    54→}

    56→function calcTitleLayout(title, maxWidthTwips, preferredPt = 40, minPt = 24) {
    57→  // English-heavy title: Latin chars ~ pt*11 twips wide
    58→  const estWidth = (text, pt) => {
    59→    let w = 0;
    60→    for (const ch of text) {
    61→      const code = ch.codePointAt(0);
    62→      const isCJK = (code >= 0x4E00 && code <= 0x9FFF) || (code >= 0x3000 && code <= 0x303F);
    63→      w += isCJK ? pt * 20 : pt * 11;
    64→    }
    65→    return w;
    66→  };
    67→  let titlePt = preferredPt, lines;
    68→  while (titlePt >= minPt) {
    69→    const avgChar = estWidth(title, titlePt) / title.length;
    70→    const cpl = Math.max(2, Math.floor(maxWidthTwips / avgChar));
    71→    lines = splitTitleLines(title, cpl);
    72→    if (lines.length <= 3) break;
    73→    titlePt -= 2;
    74→  }
    75→  if (!lines || lines.length > 3) {
    76→    const avgChar = estWidth(title, minPt) / title.length;
    77→    lines = splitTitleLines(title, Math.max(2, Math.floor(maxWidthTwips / avgChar)));
    78→    titlePt = minPt;
    79→  }
    80→  return { titlePt, titleLines: lines };
    81→}

    83→function calcCoverSpacing(params) {
    84→  const { titleLineCount = 1, titlePt = 36, hasSubtitle = false, hasEnglishLabel = false,
    85→    metaLineCount = 0, fixedHeight = 800, pageHeight = 16838 } = params;
    86→  const SAFETY = 1200;
    87→  const usableHeight = pageHeight - SAFETY;
    88→  const titleHeight = titleLineCount * (titlePt * 23 + 200);
    89→  const subtitleHeight = hasSubtitle ? (12 * 23 + 600) : 0;
    90→  const englishLabelHeight = hasEnglishLabel ? (9 * 23 + 600) : 0;
    91→  const metaHeight = metaLineCount * (10 * 23 + 100);
    92→  const implicitParaHeight = 3 * 300;
    93→  const contentHeight = titleHeight + subtitleHeight + englishLabelHeight + metaHeight + fixedHeight + implicitParaHeight;
    94→  const safeRemaining = Math.max(usableHeight - contentHeight, 400);
    95→  const FOOTER_MIN = 800;
    96→  const rawTop = Math.floor(safeRemaining * 0.45);
    97→  const rawBottom = Math.floor(safeRemaining * 0.45);
    98→  const bottomSpacing = Math.max(rawBottom, FOOTER_MIN);
    99→  const topSpacing = Math.max(rawTop - Math.max(0, FOOTER_MIN - rawBottom), 400);
   100→  return { topSpacing, midSpacing: Math.max(safeRemaining - topSpacing - bottomSpacing, 0), bottomSpacing };
   101→}

   293→bodyChildren.push(findingHead("A1", "The zero threshold of the distributional commitment gap is mischaracterized (thm:com-rd-formula)", "Major"));
   294→bodyChildren.push(locationLine("6287-6293 (statement); 6315-6327 (proof)"));
   295→bodyChildren.push(body(
   296→  "The theorem asserts that ComRD(M) = 0 if and only if M is at least the observable support index kappa_obs(F^{(1)}, mu), where F^{(1)}(u) = F(u)_{|u|} is the one-step output object. The proof equates two different determination conditions: it observes correctly that zero cost holds exactly when F(u)_{|u|} is mu-almost-surely determined by the pair (class of the prefix, next symbol), but then asserts that this is 'exactly the condition' of the class-level representative exactness that defines kappa_obs. These conditions are not equivalent. The map u -> ([u_{1:t-1}], u_t) refines the map u -> [u]: the pair determines the class, since [u] = [u_{1:t-1}] . u_t, but a single class can be reached by several distinct pairs. Determination by the coarser class map is therefore a strictly stronger requirement, and kappa_obs is an upper bound on the true threshold, not its value."
   297→));
   298→bodyChildren.push(body(
   299→  "Counterexample. Take the identity transduction F(u) = u over any nontrivial input alphabet with a full-support input process. The Myhill-Nerode index is 1 (every residual equals the identity continuation), so the one-state machine with output rule rho(*, x) = x realizes F exactly and ComRD(1) = 0. But F^{(1)}(u) = u_t is not determined by the single congruence class of the trivial congruence, since histories ending in different symbols share the class; the minimal congruence determining the last symbol has index 2 (classes: last symbol a, last symbol b). Hence ComRD(1) = 0 while kappa_obs = 2 > 1, contradicting both directions of the biconditional as stated."
   300→));
   301→bodyChildren.push(body(
   302→  "The same error undermines the secondary claim at lines 6322-6325 that kappa_obs(F^{(1)}, mu) <= kappa_det(F): the full Myhill-Nerode congruence is exact for the residual object, which determines continuations after u, not the last symbol of u itself, so the inequality fails on the same counterexample (2 > 1). There is also a small type gap: F^{(1)} is defined only on nonempty words, while mu(eps) = 1 - beta > 0 for a discounted law, so the kappa_obs instantiation is not well typed at the empty word. Recommended fix: define the threshold as the minimum index of a right congruence ~ such that F^{(1)}(u) is mu-a.e. determined by ([u_{1:|u|-1}], u_{|u|}); under that definition the residual automaton witnesses the inequality kappa_threshold <= kappa_det(F), the identity counterexample disappears, and the rest of the theorem (the greedy block formula, which is correct) is unaffected.
